    body = content.split("---", 2)[-1]
    missing = [s for s in REQUIRED_BODY_SECTIONS
               if not re.search(r"^#{2,4}[^\n]*" + s, body, re.M)]
    if missing:
        return True, ("Skill is valid! 提醒：正文缺段 %s（新建技能应补齐；骨架见 "
                      "assets/creator/init_skill.py）" % "、".join(missing))
    return True, "Skill is valid!"
