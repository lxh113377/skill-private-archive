  -h, --help            show this help message and exit
  --domain, -d {style,color,chart,landing,product,ux,typography,icons,gsap,google-fonts,react,web}

                        Max results (default: 3)
  --json                Output as JSON
  --design-system, -ds  Generate complete design system recommendation
  --project-name, -p PROJECT_NAME
                        Project name for design system output
  --format, -f {ascii,markdown}
                        Output format for design system
  --persist             Save design system to design-system/MASTER.md (creates
                        hierarchical structure)
  --page PAGE           Create page-specific override file in design-
                        system/pages/
  --output-dir, -o OUTPUT_DIR
                        Output directory for persisted files (default: current
                        directory)
